cmake_minimum_required(VERSION 3.16)
project(relic_online_tile_cube LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(RUN_MODE npu CACHE STRING "Private hardware build")
set(SOC_VERSION Ascend910B3 CACHE STRING "Explicit verified SoC")
if(NOT DEFINED ASCEND_CANN_PACKAGE_PATH)
    message(FATAL_ERROR "Pass the existing CANN root explicitly; do not install or modify CANN")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
set(RELIC_TILE_K_BLOCKS 1 CACHE STRING "Original K32 blocks consumed per native Cube tile")
if(NOT RELIC_TILE_K_BLOCKS MATCHES "^(1|2|4|8)$")
    message(FATAL_ERROR "RELIC_TILE_K_BLOCKS must preserve 1, 2, 4 or 8 original K32 blocks")
endif()
set(RELIC_TILE_OUTPUT_ROWS 16 CACHE STRING "Complete output rows per native Cube tile; source banks remain original N16")
if(NOT RELIC_TILE_OUTPUT_ROWS MATCHES "^(16|32)$")
    message(FATAL_ERROR "RELIC_TILE_OUTPUT_ROWS must be an explicit bounded 16 or 32")
endif()
set(RELIC_TILE_PRELOAD_A 0 CACHE STRING "Reuse unchanged activation in bounded L1 storage")
if(NOT RELIC_TILE_PRELOAD_A MATCHES "^(0|1)$")
    message(FATAL_ERROR "RELIC_TILE_PRELOAD_A must be an explicit 0 or 1")
endif()
set(RELIC_TILE_DIRECT_BITS 0 CACHE STRING "Construct original scaled BF16 bits without floating weight arithmetic")
if(NOT RELIC_TILE_DIRECT_BITS MATCHES "^(0|1)$")
    message(FATAL_ERROR "RELIC_TILE_DIRECT_BITS must be an explicit 0 or 1")
endif()
set(RELIC_TILE_PREPARED_TABLES 0 CACHE STRING "Copy immutable decode lookup/index tables instead of rebuilding them per launch")
if(NOT RELIC_TILE_PREPARED_TABLES MATCHES "^(0|1)$")
    message(FATAL_ERROR "RELIC_TILE_PREPARED_TABLES must be an explicit 0 or 1")
endif()
# One generated source entry embeds the geometry before including the kernel.
# Both the host workspace query and the device compiler see the same literal.
configure_file(kernel_entry.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/online_tile_cube_entry.cpp @ONLY)
ascendc_library(relic_online_tile_cube SHARED ${CMAKE_CURRENT_BINARY_DIR}/online_tile_cube_entry.cpp)
